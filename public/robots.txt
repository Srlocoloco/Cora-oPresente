# principalmente, indica onde está o sitemap.
#
# O "npm run build" copia esta pasta (public/) para dentro de dist/, que é o
# que vai publicado — não precisa subir este arquivo à mão.

Disallow: /*?verificar-email=
Disallow: /*?redefinir-senha=

# ?origem=app é o atalho do app instalado no celular (PWA): abre a mesma home,
